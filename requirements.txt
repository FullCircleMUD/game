# Evennia MUD framework.
#
# Pinned rather than floating: evennia-shards patches Evennia internals
# (at_post_login, create_character, WebSocketClient.onOpen) and keeps a
# register of every patch to re-diff on upgrade. A floating version would
# make that upgrade happen during a Railway rebuild instead of during a
# deliberate test run. 6.1.0 is the version the library suites pass against.
# Bumping is a decision, not a side effect — re-diff the register first:
# libraries/evennia-shards/docs/library-integration-risks.md
evennia==6.1.0

# LLM integration (OpenRouter API, OpenAI-compatible + embeddings)
openai

# XRPL blockchain integration (transactions, AMM, wallet auth)
xrpl-py

# Vector memory cosine similarity (ai_memory app)
numpy

# pgvector Django integration (vector search on PostgreSQL).
#
# Pinned because this is only half a version pair — the other half is the
# `vector` extension installed in the database itself. This package supplies
# VectorField and the distance functions; floating it means a fresh install
# can differ from the client tested against that server version, and from
# what every other deployment is running.
pgvector==0.5.0

# needed for xaman API
requests

# PostgreSQL driver (production — Railway)
psycopg2-binary

# Parse DATABASE_URL into Django DATABASES format
dj-database-url

# Bundled SQLite, macOS only. Apple's /usr/lib/libsqlite3.dylib runs
# sqlite3_initialize() through libdispatch, which cannot survive the fork
# twistd uses to daemonize — so `evennia start` hangs silently on the
# child's first SQLite call. settings.py swaps this in ahead of any
# sqlite3 import. The marker means pip skips it on Linux (Railway runs
# Postgres and never loads SQLite at all).
# See libraries/evennia-shards/docs/deployment-topology.md § macOS.
sqlean.py; sys_platform == "darwin"

# ── FCM libraries ──────────────────────────────────────────────────
# Installed from PyPI, not from ../libraries/. Railway clones only this
# repo, so a relative path to a sibling checkout does not resolve there.
# Pinned so all three deployed services build from identical library code.
# evennia-yaml-reader arrives transitively via the two builders.

# Declarative YAML world authoring — the wb_build admin command.
evennia-world-builder==0.1.0

# Declarative YAML mob spawning — the ms_load admin command.
evennia-mob-spawner==0.1.1

# Split-deployment / horizontal sharding. Dormant in monolith mode;
# activated by SHARDS_ROLE in settings_router.py / settings_shard0.py.
#
# 0.1.3 refuses any ObjectDB INSERT that would land with shard_id=NULL —
# a row owned by no shard, invisible to every shard's auto-filter. Two
# things in this repo depend on that version specifically:
#   - at_post_login wraps the account-bank insert in the library's
#     allow_unstamped_insert(), which does not exist before 0.1.3.
#   - A fresh database must boot shard0 before the router. The router is
#     unscoped, so its inserts now fail rather than quietly writing NULL
#     rows for Limbo and the superuser.
# It also routes all library logging to shards.log under LOG_DIR.
#
# 0.1.2 adds SHARDS_TICKET_BIND_IP — see settings_common_shard_config.py,
# which sets it False because no process on Railway can observe the
# player's address. 0.1.1 was the first wheel to carry the library's
# static assets; 0.1.0 shipped Python only, so shard_redirect.js 404'd
# and cross-shard redirects were received but never acted on.
evennia-shards==0.1.3
